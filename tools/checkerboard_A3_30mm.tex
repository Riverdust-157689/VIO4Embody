\documentclass{article}
\usepackage[a3paper, landscape, margin=5mm]{geometry}
\usepackage{tikz}
\pagestyle{empty}
\setlength{\parindent}{0pt}
\begin{document}
\begin{center}
\begin{tikzpicture}
  \def\s{30mm} % 方格边长
  \foreach \row in {0,...,8}{
    \foreach \col in {0,...,11}{
      \pgfmathparse{int(mod(\row+\col,2))}
      \ifnum\pgfmathresult=0
        \fill[black] (\col*\s,-\row*\s) rectangle ++(\s,\s);
      \fi
    }
  }
\end{tikzpicture}
